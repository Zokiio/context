func TestReviewProcessWriter(t *testing.T) {
  role := os.Getenv("CTX_REVIEW_ROLE")
  if role == "" { t.Skip("subprocess helper") }
  root := os.Getenv("CTX_REVIEW_ROOT")
  request := SetupRequest{Cwd:filepath.Join(root,"checkout"), Home:filepath.Join(root,"home"), Records:filepath.Join(root,"records")}
  other := "personal"
  if role == "personal" { request.Personal = true; request.AllowSources = []string{request.Cwd}; other = "shared" }
  plan, err := PrepareSetup(context.Background(), request)
  if err != nil { t.Fatal(err) }
  ops := defaultSetupWriteOps()
  ops.rename = func(from, to string) error {
    if err := os.WriteFile(filepath.Join(root,role+"-ready"),nil,0600); err != nil { return err }
    until := time.Now().Add(5*time.Second)
    for {
      if _, err := os.Stat(filepath.Join(root,other+"-ready")); err == nil { break }
      if time.Now().After(until) { t.Fatal("other process did not reach commit barrier") }
      time.Sleep(time.Millisecond)
    }
    return os.Rename(from,to)
  }
  if err := plan.apply(context.Background(),ops); err != nil { t.Fatal(err) }
  t.Log("setup successfully committed",role)
}

func TestReviewConcurrentSetupProcesses(t *testing.T) {
  request,_ := setupWriteFixture(t,false)
  root := filepath.Dir(request.Home)
  results := make(chan struct{ role, output string; err error },2)
  for _,role := range []string{"shared","personal"} {
    go func(role string) {
      command := exec.Command(os.Args[0],"-test.run=^TestReviewProcessWriter$","-test.v")
      command.Env = append(os.Environ(),"CTX_REVIEW_ROLE="+role,"CTX_REVIEW_ROOT="+root)
      output,err := command.CombinedOutput()
      results <- struct{ role, output string; err error }{role,string(output),err}
    }(role)
  }
  for range 2 {
    result := <-results
    t.Logf("%s writer exit=%v\n%s",result.role,result.err,result.output)
    if result.err != nil { t.Fatal("writer process failed") }
  }
  _,err := Resolve(context.Background(),Request{Cwd:request.Cwd,Home:request.Home})
  t.Logf("after both processes committed: %v",err)
  if err != nil && strings.Contains(err.Error(),"conflicting") { t.Fatal("both local processes succeeded and left conflicting declarations") }
}
